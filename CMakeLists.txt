cmake_minimum_required(VERSION 3.21)
project(iLightDesktop VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
if(MSVC)
    add_compile_options(/utf-8 /W4 /permissive-)
    add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
# This machine uses Qt5_DIR as an installation prefix (vcpkg-config convention).
if(NOT Qt5_DIR AND DEFINED ENV{Qt5_DIR})
    set(Qt5_DIR "$ENV{Qt5_DIR}/lib/cmake/Qt5")
endif()
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Widgets Bluetooth)

add_library(ilight_protocol src/protocol.cpp src/session.cpp src/session.h)
target_include_directories(ilight_protocol PUBLIC src)
target_link_libraries(ilight_protocol PUBLIC Qt5::Core)
add_executable(iLightDesktop WIN32 src/main.cpp src/maindialog.cpp src/maindialog.h src/bluetoothclient.cpp src/bluetoothclient.h)
target_link_libraries(iLightDesktop PRIVATE ilight_protocol Qt5::Widgets Qt5::Bluetooth)
if(WIN32)
    target_sources(iLightDesktop PRIVATE src/nativerfcomm.cpp src/nativerfcomm.h)
    target_sources(iLightDesktop PRIVATE src/bleadvertisements.cpp src/bleadvertisements.h)
    target_sources(iLightDesktop PRIVATE src/nativeble.cpp src/nativeble.h)
    target_sources(iLightDesktop PRIVATE src/nativeblescanner.cpp src/nativeblescanner.h)
    target_sources(iLightDesktop PRIVATE src/nativeaudio.cpp src/nativeaudio.h)
    if(MSVC)
        set_source_files_properties(src/bleadvertisements.cpp src/nativeble.cpp src/nativeblescanner.cpp PROPERTIES COMPILE_OPTIONS "/await")
    endif()
    target_link_libraries(iLightDesktop PRIVATE ws2_32 windowsapp bthprops)
endif()

install(TARGETS iLightDesktop RUNTIME DESTINATION .)
